cmake_minimum_required(VERSION 3.16)
project(elpis_ecs_memory_experimental C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(elpis_memory_experimental STATIC
    src/bounded_lattice.c src/indexed_lattice.c src/model_free_memory.c)
target_include_directories(elpis_memory_experimental PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include/elpis)
target_link_libraries(elpis_memory_experimental PUBLIC m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(elpis_memory_experimental PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
if(BUILD_TESTING)
    add_executable(test_model_free tests/test_model_free.c)
    target_link_libraries(test_model_free PRIVATE elpis_memory_experimental)
    add_test(NAME model_free_adapter COMMAND test_model_free)
    if(CMAKE_C_COMPILER_ID STREQUAL "GNU" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(test_index tests/test_index.c)
        target_link_libraries(test_index PRIVATE elpis_memory_experimental)
        target_link_options(test_index PRIVATE -Wl,--wrap=calloc -Wl,--wrap=malloc -Wl,--wrap=free)
        add_test(NAME bounded_vs_indexed_differential COMMAND test_index)
    endif()
endif()
